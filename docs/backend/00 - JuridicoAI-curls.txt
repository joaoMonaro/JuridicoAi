# JuridicoAI — 42 endpoints
# Execute os comandos individualmente e substitua as variáveis {{...}} pelos valores retornados pela API.

# 00 - Infraestrutura
curl --request GET --url 'http://localhost:5299/' --header 'Accept: text/plain'

# 01 - Autenticação
curl --request POST --url 'http://localhost:5299/auth/register' \
  --header 'Content-Type: application/json' \
  --cookie-jar '/tmp/juridicoai-cookies.txt' \
  --data '{"organizationName":"Escritório Exemplo","name":"João da Silva","email":"joao@example.com","password":"senha-segura-com-12-ou-mais"}'

curl --request POST --url 'http://localhost:5299/auth/login' \
  --header 'Content-Type: application/json' \
  --cookie-jar '/tmp/juridicoai-cookies.txt' \
  --data '{"email":"joao@example.com","senha":"senha-segura-com-12-ou-mais"}'

curl --request POST --url 'http://localhost:5299/auth/refresh' \
  --cookie '/tmp/juridicoai-cookies.txt' --cookie-jar '/tmp/juridicoai-cookies.txt'

curl --request GET --url 'http://localhost:5299/auth/me' \
  --header 'Authorization: Bearer {{accessToken}}'

# 02 - Casos
curl --request POST --url 'http://localhost:5299/casos' \
  --header 'Authorization: Bearer {{accessToken}}' --header 'Content-Type: application/json' \
  --data '{"titulo":"Negativação indevida - João da Silva","descricao":"Cliente relata dívida desconhecida."}'

curl --request GET --url 'http://localhost:5299/casos?status=Ativo' \
  --header 'Authorization: Bearer {{accessToken}}'

curl --request GET --url 'http://localhost:5299/casos/{{casoId}}' \
  --header 'Authorization: Bearer {{accessToken}}'

curl --request PUT --url 'http://localhost:5299/casos/{{casoId}}' \
  --header 'Authorization: Bearer {{accessToken}}' --header 'Content-Type: application/json' \
  --data '{"titulo":"Negativação indevida - João da Silva","descricao":"Relato revisado.","status":"Ativo"}'

curl --request DELETE --url 'http://localhost:5299/casos/{{casoId}}' \
  --header 'Authorization: Bearer {{accessToken}}'
